\subsection{Functional Requirements}
\textbf{Farmer Requirements}
\begin{itemize}
    \item User registration
    \item Login authentication
    \item Create pickup requests
    \item View pickup history
    \item Track request status
\end{itemize}
\textbf{Collector Requirements}
\begin{itemize}
    \item Login authentication
    \item View pickup requests
    \item Accept requests
    \item Record actual quantity
    \item Complete requests
\end{itemize}
