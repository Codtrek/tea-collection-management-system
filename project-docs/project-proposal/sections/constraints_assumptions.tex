\section{Project Constraints and Assumptions}
\label{sec:constraints_assumptions}

\subsection{Constraints}
\begin{itemize}
    \item Internet connectivity issues in rural areas
    \item Limited technical knowledge of some users
    \item Limited project timeline
\end{itemize}

\subsection{Assumptions}
\begin{itemize}
    \item Users have Android smartphones
    \item Users can access mobile internet
    \item Collectors are willing to adopt digital systems
\end{itemize}
