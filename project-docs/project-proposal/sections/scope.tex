\section{Scope of the Project}
\label{sec:scope}

\subsection{Main Users}
\begin{itemize}
    \item Farmers
    \item Collectors
    \item Collection Centers
\end{itemize}

\subsection{In-Scope Functionalities}
\textbf{Farmer Functionalities}
\begin{itemize}
    \item Register/Login
    \item Create pickup requests
    \item Enter product quantity
    \item Track request status
    \item View collection history
\end{itemize}
\textbf{Collector Functionalities}
\begin{itemize}
    \item Login
    \item View pending pickup requests
    \item Accept pickup requests
    \item Enter actual measured quantity
    \item Mark requests as completed
    \item View collection history
\end{itemize}
\textbf{Shared Functionalities}
\begin{itemize}
    \item Digital record keeping
    \item Timestamp tracking
    \item Transparent transaction history
\end{itemize}

\subsection{Out of Scope}
\begin{itemize}
    \item Online payment system
    \item AI-based predictions
    \item Advanced analytics
    \item Export management systems
    \item Multi-country support
\end{itemize}

\subsection{Flow of Events in Current Context}
\begin{enumerate}
    \item Farmer collects agricultural products.
    \item Farmer contacts collector manually.
    \item Farmer informs quantity and location.
    \item Collector visits location.
    \item Collector measures products.
    \item Both parties record information manually in books.
\end{enumerate}
Problems:
\begin{itemize}
    \item Duplicate manual records
    \item Human errors
    \item Time consumption
    \item No centralized system
\end{itemize}

\subsection{Flow of Events with Proposed System}
\begin{enumerate}
    \item Farmer creates pickup request through mobile app.
    \item Collector receives notification/request.
    \item Collector accepts request.
    \item Collector visits location and measures products.
    \item Collector records actual quantity digitally.
    \item Both parties can view shared records instantly.
\end{enumerate}
